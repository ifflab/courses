# Read the windows: xi samples, window centre xi0 and force constant k windows = [np.load(f) for f in sys.argv[1:]] samples = [w["xi"] for w in windows] xi0 = np.array([w["xi0"] for w in windows]) k = np.array([w["k"] for w in windows]) # Histogram of each window on common bins covering all samples all_xi = np.concatenate(samples) bins = np.arange(all_xi.min(), all_xi.max() + BIN_WIDTH, BIN_WIDTH) xi = 0.5 * (bins[1:] + bins[:-1]) # bin centres hist = np.array([histogram(s, bins) for s in samples]) sampled = hist.sum(axis=0) > 0 # drop empty bins xi, hist = xi[sampled], hist[:, sampled] # Free-energy profile, then barrier and reaction free energy relative to the reactant minimum F = wham(hist, xi, xi0, k) ts = np.argmax(F) # transition state: highest point F_reactant, F_product = F[:ts].min(), F[ts:].min() # lowest points before and after it print(f"barrier = {F[ts] - F_reactant:.1f} kcal/mol, reaction free energy = {F_product - F_reactant:.1f} kcal/mol") # Save the profile as a table and as a plot np.savetxt("pmf.dat", np.column_stack([xi, F]), fmt="%.3f", header="xi (A) F (kcal/mol)") plt.plot(xi, F) plt.xlabel("xi (A)") plt.ylabel("F (kcal/mol)") plt.savefig("pmf.png", dpi=150)
