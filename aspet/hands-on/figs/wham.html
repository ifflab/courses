def wham(hist, xi, xi0, k): """Free energy (kcal/mol, minimum = 0) at bin centres xi from the window histograms (window x bin) of umbrellas with centres xi0 and force constants k.""" bias = np.exp(-0.5 * k[:, None] * (xi - xi0[:, None]) ** 2 / KT) # Boltzmann factor of each umbrella N = hist.sum(axis=1) Z = np.ones(len(hist)) # window partition functions for _ in range(10000): P = hist.sum(axis=0) / ((N / Z) @ bias) Z = bias @ P F = -KT * np.log(P) return F - F.min()
