us_window.py
import os
import sys
import numpy as np
import openmm as mm
import openmm.unit as u
import sire as sr
from emle.calculator import EMLECalculator

K_US = 300 * u.kilocalorie_per_mole / u.angstrom**2     # umbrella force constant
K_WALL = 600 * u.kilocalorie_per_mole / u.angstrom**2   # wall force constant
R_WALL = 3.0 * u.angstrom                               # walls keep both C-Cl distances below this
T = 300 * u.kelvin

win = int(sys.argv[1])                                  # window index, 0 ... 28

# Solute (molecule 0) with AIMNet2 + EMLE, water with TIP3P, starting from the pre-relaxed structure
mols = sr.load("box.prm7", f"start/win_{win:02d}.rst7")
calc = EMLECalculator(model="sn2_model.mat", backend="aimnet2", alpha_mode="flexible", qm_charge=-1)
qm_mols, engine = sr.qm.emle(mols, mols[0], calc, cutoff="12A")
context = qm_mols.dynamics(temperature="300K", qm_engine=engine, platform="CPU").context()

Reaction coordinate and bias

us_window.py
# Reaction coordinate: atoms C (0), Cl1 (4), Cl2 (5)
xi = mm.CustomCompoundBondForce(3, "distance(p1,p2)-distance(p1,p3)")
xi.addBond([0, 4, 5])

# Window centre: xi0 = -1.4, -1.3, ..., +1.4 A
xi0_A = -1.4 + 0.1 * win
xi0 = xi0_A * u.angstrom

# Umbrella on xi and walls on both C-Cl
umbrella = mm.CustomCVForce("0.5*k*(xi-xi0)^2")
umbrella.addCollectiveVariable("xi", xi)
umbrella.addGlobalParameter("k", K_US)
umbrella.addGlobalParameter("xi0", xi0)
wall = mm.CustomBondForce("step(r-r_wall)*0.5*k_wall*(r-r_wall)^2")
wall.addGlobalParameter("k_wall", K_WALL)
wall.addGlobalParameter("r_wall", R_WALL)
wall.addBond(0, 4)
wall.addBond(0, 5)
context.getSystem().addForce(umbrella)
context.getSystem().addForce(wall)
context.reinitialize(preserveState=True)

pmf.py
"""Free-energy profile F(xi) with WHAM from the umbrella windows given on the command line.
Usage: python pmf.py us/*.npz   ->  pmf.dat, pmf.png"""
import sys
import numpy as np
import matplotlib.pyplot as plt

KT = 0.0019872 * 300.0                          # kcal/mol
BIN_WIDTH = 0.05                                # A; a window spreads over sd = sqrt(KT/k) = 0.045 A at k = 300

def histogram(xi_values, bins):
    """Number of xi samples of one window in each bin"""
    counts, _ = np.histogram(xi_values, bins)
    return counts
Common bins of 0.05 Å, about one standard deviation of a window Neighbouring windows must overlap, or WHAM cannot connect them
Step 4 · Analysis · pmf.py 2/3

WHAM: remove the bias, combine the windows

pmf.py
def wham(hist, xi, xi0, k):
    """Free energy (kcal/mol, minimum = 0) at bin centres xi from the window histograms
    (window x bin) of umbrellas with centres xi0 and force constants k."""
    bias = np.exp(-0.5 * k[:, None] * (xi - xi0[:, None]) ** 2 / KT)   # Boltzmann factor of each umbrella
    N = hist.sum(axis=1)                        # number of samples in each window
    Z = np.ones(len(hist))                      # window partition functions
    # WHAM equations, solved by iteration: the unbiased probability P of each bin combines all windows,
    # each window weighted by its number of samples and its bias; Z then follows from P
    for _ in range(10000):
        P = hist.sum(axis=0) / ((N / Z) @ bias)
        Z = bias @ P
    F = -KT * np.log(P)                         # free energy from probability
    return F - F.min()
$c_{ij} = e^{-k_i(\xi_j-\xi_{0,i})^2/2k_BT}$bias factor of
window i in bin j
$P_j = \dfrac{\sum_i n_{ij}}{\sum_i N_i\,c_{ij}/Z_i}$unbiased
probability
$Z_i = \sum_j c_{ij}\,P_j$partition function
of window i
↻ P needs Z, Z needs P: iterate to self-consistency
$F_j = -k_BT\ln P_j$free energy
i: window, j: bin, $n_{ij}$: hist, $N_i$: samples in window i
Step 4 · Analysis · pmf.py 3/3

Run it: F(ξ), barrier, plot

pmf.py
# Read the windows: xi samples, window centre xi0 and force constant k
windows = [np.load(f) for f in sys.argv[1:]]
samples = [w["xi"] for w in windows]
xi0 = np.array([w["xi0"] for w in windows])
k = np.array([w["k"] for w in windows])

# Histogram of each window on common bins covering all samples
all_xi = np.concatenate(samples)
bins = np.arange(all_xi.min(), all_xi.max() + BIN_WIDTH, BIN_WIDTH)
xi = 0.5 * (bins[1:] + bins[:-1])              # bin centres
hist = np.array([histogram(s, bins) for s in samples])
sampled = hist.sum(axis=0) > 0                  # drop empty bins
xi, hist = xi[sampled], hist[:, sampled]

# Free-energy profile, then barrier and reaction free energy relative to the reactant minimum
F = wham(hist, xi, xi0, k)
ts = np.argmax(F)                               # transition state: highest point
F_reactant, F_product = F[:ts].min(), F[ts:].min()   # lowest points before and after it
print(f"barrier = {F[ts] - F_reactant:.1f} kcal/mol, reaction free energy = {F_product - F_reactant:.1f} kcal/mol")

# Save the profile as a table and as a plot
np.savetxt("pmf.dat", np.column_stack([xi, F]), fmt="%.3f", header="xi (A)  F (kcal/mol)")
plt.plot(xi, F)
plt.xlabel("xi (A)")
plt.ylabel("F (kcal/mol)")
plt.savefig("pmf.png", dpi=150)
